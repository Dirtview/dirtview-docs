{/* Trims a URL at the first of any of the given characters. */}
export const trimAt = (value, stops) => {
  let out = value;
  for (const stop of stops) {
    const at = out.indexOf(stop);
    if (at !== -1) out = out.slice(0, at);
  }
  return out;
};

{/* Turns a normal share link into an embeddable one. */}
export const toEmbedUrl = (url) => {
  if (!url) return "";
  if (url.includes("loom.com/share/")) {
    return trimAt(url, ["?", "#"]).replace("/share/", "/embed/");
  }
  if (url.includes("youtu.be/")) {
    return "https://www.youtube.com/embed/" + trimAt(url.split("youtu.be/")[1], ["?", "&", "#", "/"]);
  }
  if (url.includes("watch?v=")) {
    return "https://www.youtube.com/embed/" + trimAt(url.split("watch?v=")[1], ["&", "#"]);
  }
  if (url.includes("vimeo.com/") && !url.includes("player.vimeo.com")) {
    return "https://player.vimeo.com/video/" + trimAt(url.split("vimeo.com/")[1], ["?", "&", "#", "/"]);
  }
  return url;
};
